3//! DICE layer 0 for the CMVM monitor (F0X-80).
4//!
5//! The monitor's Compound Device Identifier comes from a VMPL0 key the PSP
6//! derives from the VCEK, mixed with this guest's launch measurement, guest
7//! policy and reported TCB. The same SVSM image, launched with the same policy
8//! on the same chip at the same TCB, gets the same CDI; any of those changing
9//! gives a different one. The VMRK is not used: it changes on every launch.
10//!
11//! The alias key is derived from the CDI the same way `cmvm-dice` derives it
12//! in uPolaris, and bound into a VMPL0 attestation report as
13//! `REPORT_DATA = SHA-512(REPORT_DATA_TAG || SEC1 point)`. That report, signed
14//! by the VCEK, is the hardware root endorsement: it says this measurement
15//! holds this alias key.
16//!
17//! The CDI and the alias private key never leave VMPL0.

40const INFO_MONITOR_CDI: &[u8] = b"cmvm monitor cdi v1";
41const INFO_PARTITION_CDI: &[u8] = b"cmvm partition cdi v1";
42const INFO_ALIAS: &[u8] = b"cmvm alias v1";
43
44/// Domain tag hashed in front of the alias public key to form REPORT_DATA.
45pub const REPORT_DATA_TAG: &[u8] = b"cmvm monitor alias v1\0";
46
47/// Size of the uncompressed SEC1 encoding of a P-384 public key.
48pub const ALIAS_PUBLIC_SIZE: usize = 97;
49
50/// Size of a partition wrap public key: an uncompressed SEC1 P-384 point.
51pub const WRAP_PUBLIC_SIZE: usize = 97;
52
53/// Size of an SNP attestation report.
54pub const REPORT_SIZE: usize = size_of::<AttestationReport>();
55
56/// The guest fields the monitor CDI is bound to.
57const CDI_FIELDS: u64 = FIELD_MEASUREMENT | FIELD_GUEST_POLICY | FIELD_TCB_VERSION;
58
59/// A VMPL0 attestation report carrying `report_data`.
60fn vmpl0_report(report_data: [u8; 64]) -> Result<AttestationReport, SvsmError> {
61    let mut buffer = [0u8; size_of::<SnpReportResponse>()];
62    let request = SnpReportRequest::new(report_data, 0, 0);
63    buffer[..size_of::<SnpReportRequest>()].copy_from_slice(request.as_bytes());
64    get_regular_report(&mut buffer)?;
65    let (response, _rest) =
66        SnpReportResponse::ref_from_prefix(&buffer).map_err(|_| SvsmError::InvalidParameter)?;
67    Ok(*response.report())
68}

70/// The monitor CDI: HKDF-SHA384 over the VCEK-rooted derived key.
71fn monitor_cdi() -> Result<Zeroizing<[u8; 48]>, SvsmError> {
72    Ok(monitor_cdi_and_measurement()?.0)
73}
74
75/// The monitor CDI and the monitor's launch measurement, read from the same
76/// VMPL0 report.
77fn monitor_cdi_and_measurement() -> Result<(Zeroizing<[u8; 48]>, [u8; 48]), SvsmError> {
78    // TCB_VERSION mixes in the value the request carries, so pass the TCB the
79    // VCEK is derived at, read from this guest's own report. It is never above
80    // the committed TCB, which the PSP requires.
81    let own = vmpl0_report([0u8; 64])?;
82    let tcb = own.reported_tcb();
83    let request = SnpDerivedKeyRequest::new(ROOT_KEY_VCEK, CDI_FIELDS, 0, tcb)?;
84    let key = get_derived_key(&request)?;
85    let mut cdi = Zeroizing::new([0u8; 48]);
86    Hkdf::<Sha384>::new(None, &key[..])
87        .expand(INFO_MONITOR_CDI, &mut cdi[..])
88        .map_err(|_| SvsmError::InvalidParameter)?;
89    Ok((cdi, *own.measurement()))
90}

92/// A partition's CDI: HKDF-SHA384 with the monitor CDI as input keying
93/// material and the partition measurement as salt, as `cmvm-dice` derives it.
94fn partition_cdi(monitor: &[u8; 48], measurement: &[u8]) -> Result<Zeroizing<[u8; 48]>, SvsmError> {
95    let mut cdi = Zeroizing::new([0u8; 48]);
96    Hkdf::<Sha384>::new(Some(measurement), monitor)
97        .expand(INFO_PARTITION_CDI, &mut cdi[..])
98        .map_err(|_| SvsmError::InvalidParameter)?;
99    Ok(cdi)
100}

102fn sec1(key: &SigningKey) -> Result<[u8; ALIAS_PUBLIC_SIZE], SvsmError> {
103    key.verifying_key()
104        .to_sec1_point(false)
105        .as_bytes()
106        .try_into()
107        .map_err(|_| SvsmError::InvalidFormat)
108}

110/// Check a partition wrap public key before it goes into a signed token: an
111/// uncompressed SEC1 point on P-384 that is not the identity. The SVSM encodes
112/// the COSE_Key itself from the checked point, so the guest never puts raw
113/// CBOR into a token the SVSM signs.
114pub fn check_wrap_key(sec1: &[u8; WRAP_PUBLIC_SIZE]) -> Result<(), SvsmError> {
115    if sec1[0] != 0x04 {
116        return Err(SvsmError::InvalidFormat);
117    }
118    VerifyingKey::from_sec1_bytes(sec1)
119        .map(|_| ())
120        .map_err(|_| SvsmError::InvalidFormat)
121}

123/// Measure, derive and sign for one partition: the monitor token, carrying the
124/// partition's nested token and the monitor's endorsement of its measurement
125/// and alias key. With [`root_endorsement`] it is a complete version 2
126/// evidence bundle.
127///
128/// With `wrap_key`, the nested token also carries the partition's wrap key in
129/// its `cnf` claim (RFC 8747, F0X-898), so the token names the key a KMS
130/// releases to and not only its hash in `report_data`. `report_data` is still
131/// the caller's, unchecked here: a verifier requires it to be
132/// `SHA-512(nonce || COSE_Key)` for the same key.
133///
134/// Prototype limit: the nested token is signed here, in VMPL0, with the alias
135/// key derived for `partition_measurement`. In the eventual design the
136/// partition holds its own key and signs it; where it does depends on how
137/// partitions run (F0X-76).
138pub fn attest_partition(
139    nonce: &[u8],
140    report_data: &[u8],
141    partition_id: &str,
142    partition_measurement: &[u8; 48],
143    wrap_key: Option<&[u8; WRAP_PUBLIC_SIZE]>,
144) -> Result<Vec<u8>, SvsmError> {
145    if let Some(w) = wrap_key {
146        check_wrap_key(w)?;
147    }
148    let (cdi, monitor_measurement) = monitor_cdi_and_measurement()?;
149    let monitor_alias = alias_key(&cdi)?;
150    let partition = partition_cdi(&cdi, partition_measurement)?;
151    let partition_alias = alias_key(&partition)?;
152    let nested = eat::nested_token(
153        &partition_alias,
154        nonce,
155        partition_measurement,
156        report_data,
157        wrap_key,
158    );
159    Ok(eat::monitor_token(
160        &monitor_alias,
161        nonce,
162        &monitor_measurement,
163        partition_id,
164        partition_measurement,
165        &sec1(&partition_alias)?,
166        &nested,
167    ))
168}

170/// The alias key for a CDI. Re-draws with a counter in the negligible case the
171/// derived bytes are not a valid P-384 scalar, as `cmvm-dice` does.
172fn alias_key(cdi: &[u8; 48]) -> Result<SigningKey, SvsmError> {
173    let hk = Hkdf::<Sha384>::new(None, cdi);
174    for ctr in 0u8..=u8::MAX {
175        let mut info = Vec::from(INFO_ALIAS);
176        if ctr > 0 {
177            info.push(ctr);
178        }
179        let mut okm = Zeroizing::new([0u8; 48]);
180        hk.expand(&info, &mut okm[..])
181            .map_err(|_| SvsmError::InvalidParameter)?;
182        if let Ok(key) = SigningKey::from_slice(&okm[..]) {
183            return Ok(key);
184        }
185    }
186    Err(SvsmError::InvalidParameter)
187}

189/// `REPORT_DATA` for an alias public key.
190pub fn alias_report_data(alias_public: &[u8]) -> [u8; 64] {
191    let mut h = Sha512::new();
192    h.update(REPORT_DATA_TAG);
193    h.update(alias_public);
194    h.finalize().into()
195}
196
197/// The hardware root endorsement: the monitor's alias public key and a VMPL0
198/// report binding it. The verifier recomputes `REPORT_DATA` from the key and
199/// checks the report against the VCEK chain.
200pub fn root_endorsement() -> Result<([u8; ALIAS_PUBLIC_SIZE], AttestationReport), SvsmError> {
201    let cdi = monitor_cdi()?;
202    let public = sec1(&alias_key(&cdi)?)?;
203    let report = vmpl0_report(alias_report_data(&public))?;
204    Ok((public, report))
205}

207#[cfg(test)]
208mod tests {
209    use super::*;
210
211    #[test]
212    fn alias_key_is_deterministic_in_the_cdi() {
213        let a = alias_key(&[7u8; 48]).unwrap();
214        let b = alias_key(&[7u8; 48]).unwrap();
215        assert_eq!(a.verifying_key(), b.verifying_key());
216        assert_ne!(
217            a.verifying_key(),
218            alias_key(&[8u8; 48]).unwrap().verifying_key()
219        );
220    }
221
222    #[test]
223    fn check_wrap_key_takes_only_uncompressed_points_on_p384() {
224        let good = SigningKey::from_slice(&[7u8; 48])
225            .unwrap()
226            .verifying_key()
227            .to_sec1_point(false);
228        let good: [u8; WRAP_PUBLIC_SIZE] = good.as_bytes().try_into().unwrap();
229        assert!(check_wrap_key(&good).is_ok());
230        let mut off_curve = good;
231        off_curve[WRAP_PUBLIC_SIZE - 1] ^= 1;
232        assert!(check_wrap_key(&off_curve).is_err());
233        let mut wrong_tag = good;
234        wrong_tag[0] = 0x02;
235        assert!(check_wrap_key(&wrong_tag).is_err());
236        assert!(check_wrap_key(&[0u8; WRAP_PUBLIC_SIZE]).is_err());
237    }
238
239    #[test]
240    fn report_data_binds_the_whole_key() {
241        let k = [1u8; ALIAS_PUBLIC_SIZE];
242        let mut other = k;
243        other[ALIAS_PUBLIC_SIZE - 1] ^= 1;
244        assert_ne!(alias_report_data(&k), alias_report_data(&other));
245    }
