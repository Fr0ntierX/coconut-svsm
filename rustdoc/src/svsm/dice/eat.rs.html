3//! The CMVM EAT tokens (F0X-80), built and signed in VMPL0.
4//!
5//! The layout is the `tag:fr0ntierx.com,2026:cmvm-eat-v1` profile that
6//! `cmvm-dice` in uPolaris defines and `cmvm-verify` appraises: a monitor
7//! token signed by the monitor alias key, carrying the nonce, the monitor
8//! measurement, each partition's nested token under `submods`, and the
9//! monitor's endorsement of each partition's measurement and alias key. Claim
10//! keys are the RFC 9711 registrations plus the profile's private keys.
11//!
12//! Signatures are COSE_Sign1 (RFC 9052) with ES384, untagged.
13//!
14//! A nested token can also carry the partition's wrap key (F0X-898) in a CWT
15//! confirmation claim (RFC 8747), `8: {1: COSE_Key}`, byte for byte as
16//! `cmvm-dice`'s `nested_token_with_wrap_key` writes it. The wrap key is the
17//! HPKE key a KMS seals a released key to.
18

28pub const PROFILE: &str = "tag:fr0ntierx.com,2026:cmvm-eat-v1";
29
30const CLAIM_NONCE: i64 = 10;
31const CLAIM_PROFILE: i64 = 265;
32const CLAIM_SUBMODS: i64 = 266;
33const CLAIM_MEASUREMENTS: i64 = 273;
34const CLAIM_PARTITION_ENDORSEMENTS: i64 = -70001;
35const CLAIM_REPORT_DATA: i64 = -70002;
36const CLAIM_ALIAS_KEY: i64 = -70003;
37/// RFC 8747 confirmation claim, and its COSE_Key member.
38const CLAIM_CNF: i64 = 8;
39const CNF_COSE_KEY: i64 = 1;
40
41/// Measurement content type from the CoAP experimental range, for a CBOR
42/// `[alg, digest]` record. `ALG_SHA384` is the profile's tag for SHA-384.
43const CT_DIGEST: i64 = 65000;
44const ALG_SHA384: i64 = 1;
45
46/// COSE labels and values.
47const COSE_HEADER_ALG: i64 = 1;
48const COSE_ALG_ES384: i64 = -35;
49const COSE_KEY_KTY: i64 = 1;
50const COSE_KEY_ALG: i64 = 3;
51const COSE_KTY_EC2: i64 = 2;
52const COSE_EC2_CRV: i64 = -1;
53const COSE_EC2_X: i64 = -2;
54const COSE_EC2_Y: i64 = -3;
55const COSE_CRV_P384: i64 = 2;
56
57/// RFC 9711 measurements: `[[content-type, bstr .cbor [alg, digest]]]`.
58fn measurements(out: &mut Vec<u8>, digest: &[u8]) {
59    let mut record = Vec::new();
60    cbor::array(&mut record, 2);
61    cbor::int(&mut record, ALG_SHA384);
62    cbor::bytes(&mut record, digest);
63    cbor::array(out, 1);
64    cbor::array(out, 2);
65    cbor::int(out, CT_DIGEST);
66    cbor::bytes(out, &record);
67}

69/// The COSE_Key for a P-384 signing (alias) public key given as an
70/// uncompressed SEC1 point. Labeled ES384.
71pub fn cose_key(sec1: &[u8; 97]) -> Vec<u8> {
72    let mut out = Vec::new();
73    ec2_p384_key(&mut out, sec1, Some(COSE_ALG_ES384));
74    out
75}
76
77/// The COSE_Key for a P-384 wrap (HPKE) public key: no `alg`, as `cmvm-dice`
78/// `WrapPublicKey::to_cose` encodes it. Its bytes are the `wrap_pubkey` that
79/// `report_data` hashes. `cmvm-dice` refuses an ES384-labeled key as a wrap
80/// key, so an alias key can't be passed off as one.
81pub fn wrap_cose_key(sec1: &[u8; 97]) -> Vec<u8> {
82    let mut out = Vec::new();
83    ec2_p384_key(&mut out, sec1, None);
84    out
85}
86
87/// An EC2 P-384 COSE_Key in the order coset writes it: kty, alg, crv, x, y.
88fn ec2_p384_key(out: &mut Vec<u8>, sec1: &[u8; 97], alg: Option<i64>) {
89    cbor::map(out, if alg.is_some() { 5 } else { 4 });
90    cbor::int(out, COSE_KEY_KTY);
91    cbor::int(out, COSE_KTY_EC2);
92    if let Some(alg) = alg {
93        cbor::int(out, COSE_KEY_ALG);
94        cbor::int(out, alg);
95    }
96    cbor::int(out, COSE_EC2_CRV);
97    cbor::int(out, COSE_CRV_P384);
98    cbor::int(out, COSE_EC2_X);
99    cbor::bytes(out, &sec1[1..49]);
100    cbor::int(out, COSE_EC2_Y);
101    cbor::bytes(out, &sec1[49..97]);
102}

104/// An untagged COSE_Sign1 over `payload` with ES384.
105pub fn sign1(key: &SigningKey, payload: &[u8]) -> Vec<u8> {
106    let mut protected = Vec::new();
107    cbor::map(&mut protected, 1);
108    cbor::int(&mut protected, COSE_HEADER_ALG);
109    cbor::int(&mut protected, COSE_ALG_ES384);
110
111    // Sig_structure = ["Signature1", protected, external_aad, payload]
112    let mut tbs = Vec::new();
113    cbor::array(&mut tbs, 4);
114    cbor::text(&mut tbs, "Signature1");
115    cbor::bytes(&mut tbs, &protected);
116    cbor::bytes(&mut tbs, &[]);
117    cbor::bytes(&mut tbs, payload);
118    let signature: Signature = key.sign(&tbs);
119
120    let mut out = Vec::new();
121    cbor::array(&mut out, 4);
122    cbor::bytes(&mut out, &protected);
123    cbor::map(&mut out, 0);
124    cbor::bytes(&mut out, payload);
125    cbor::bytes(&mut out, &signature.to_bytes());
126    out
127}

129/// The partition's nested token claims: nonce, its measurement,
130/// `report_data` and, if given, the wrap key in `cnf`. The order is the one
131/// `cmvm-dice` writes, so the same inputs give the same bytes.
132fn nested_claims(
133    nonce: &[u8],
134    partition_measurement: &[u8],
135    report_data: &[u8],
136    wrap_key: Option<&[u8; 97]>,
137) -> Vec<u8> {
138    let mut claims = Vec::new();
139    cbor::map(&mut claims, if wrap_key.is_some() { 4 } else { 3 });
140    cbor::int(&mut claims, CLAIM_NONCE);
141    cbor::bytes(&mut claims, nonce);
142    cbor::int(&mut claims, CLAIM_MEASUREMENTS);
143    measurements(&mut claims, partition_measurement);
144    cbor::int(&mut claims, CLAIM_REPORT_DATA);
145    cbor::bytes(&mut claims, report_data);
146    if let Some(wrap) = wrap_key {
147        // The COSE_Key goes in as a CBOR map, not a byte string.
148        cbor::int(&mut claims, CLAIM_CNF);
149        cbor::map(&mut claims, 1);
150        cbor::int(&mut claims, CNF_COSE_KEY);
151        claims.extend_from_slice(&wrap_cose_key(wrap));
152    }
153    claims
154}

156/// The partition's nested token, signed by the partition alias key. With
157/// `wrap_key` (an uncompressed SEC1 P-384 point) it carries that key in
158/// `cnf`. The caller checks the point is on the curve.
159pub fn nested_token(
160    partition_alias: &SigningKey,
161    nonce: &[u8],
162    partition_measurement: &[u8],
163    report_data: &[u8],
164    wrap_key: Option<&[u8; 97]>,
165) -> Vec<u8> {
166    sign1(
167        partition_alias,
168        &nested_claims(nonce, partition_measurement, report_data, wrap_key),
169    )
170}

172/// The monitor token for one partition, signed by the monitor alias key.
173pub fn monitor_token(
174    monitor_alias: &SigningKey,
175    nonce: &[u8],
176    monitor_measurement: &[u8],
177    partition_id: &str,
178    partition_measurement: &[u8],
179    partition_alias_public: &[u8; 97],
180    nested: &[u8],
181) -> Vec<u8> {
182    let mut claims = Vec::new();
183    cbor::map(&mut claims, 5);
184    cbor::int(&mut claims, CLAIM_PROFILE);
185    cbor::text(&mut claims, PROFILE);
186    cbor::int(&mut claims, CLAIM_NONCE);
187    cbor::bytes(&mut claims, nonce);
188    cbor::int(&mut claims, CLAIM_MEASUREMENTS);
189    measurements(&mut claims, monitor_measurement);
190
191    cbor::int(&mut claims, CLAIM_SUBMODS);
192    cbor::map(&mut claims, 1);
193    cbor::text(&mut claims, partition_id);
194    cbor::bytes(&mut claims, nested);
195
196    cbor::int(&mut claims, CLAIM_PARTITION_ENDORSEMENTS);
197    cbor::map(&mut claims, 1);
198    cbor::text(&mut claims, partition_id);
199    cbor::map(&mut claims, 2);
200    cbor::int(&mut claims, CLAIM_MEASUREMENTS);
201    measurements(&mut claims, partition_measurement);
202    cbor::int(&mut claims, CLAIM_ALIAS_KEY);
203    cbor::bytes(&mut claims, &cose_key(partition_alias_public));
204
205    sign1(monitor_alias, &claims)
206}

213    fn sign1_verifies_over_the_rfc9052_sig_structure() {
214        use p384::ecdsa::signature::Verifier;
215        let key = SigningKey::from_slice(&[7u8; 48]).unwrap();
216        let msg = sign1(&key, b"payload");
217        // [bstr(protected a10138 22), {}, bstr(payload), bstr(96-byte sig)]
218        assert_eq!(&msg[..6], &[0x84, 0x44, 0xa1, 0x01, 0x38, 0x22]);
219        assert_eq!(msg[6], 0xa0);
220        let sig = Signature::from_slice(&msg[msg.len() - 96..]).unwrap();
221        let mut tbs = Vec::new();
222        cbor::array(&mut tbs, 4);
223        cbor::text(&mut tbs, "Signature1");
224        cbor::bytes(&mut tbs, &[0xa1, 0x01, 0x38, 0x22]);
225        cbor::bytes(&mut tbs, &[]);
226        cbor::bytes(&mut tbs, b"payload");
227        assert!(key.verifying_key().verify(&tbs, &sig).is_ok());
228    }

251    fn unhex(s: &str) -> Vec<u8> {
252        (0..s.len())
253            .step_by(2)
254            .map(|i| u8::from_str_radix(&s[i..i + 2], 16).unwrap())
255            .collect()
256    }
257
258    fn wrap_sec1() -> [u8; 97] {
259        unhex(WRAP_SEC1).try_into().unwrap()
260    }

262    #[test]
263    fn wrap_cose_key_matches_cmvm_dice() {
264        assert_eq!(wrap_cose_key(&wrap_sec1()), unhex(WRAP_COSE));
265        // The alias form differs only by the ES384 label: {1: 2, 3: -35, ...}.
266        let alias = cose_key(&wrap_sec1());
267        assert_eq!(&alias[..5], &[0xa5, 0x01, 0x02, 0x03, 0x38]);
268        assert_eq!(&alias[6..], &unhex(WRAP_COSE)[3..]);
269    }

271    #[test]
272    fn cnf_claim_is_rfc8747_cose_key_form() {
273        let nonce = [9u8; 32];
274        let pm = unhex(PARTITION_MEASUREMENT);
275        let rd = unhex(REPORT_DATA);
276        let with = nested_claims(&nonce, &pm, &rd, Some(&wrap_sec1()));
277        let without = nested_claims(&nonce, &pm, &rd, None);
278        // 8: {1: COSE_Key}, appended after the three claims, map head 3 -> 4.
279        let mut expected = without.clone();
280        expected[0] = 0xa4;
281        expected.extend_from_slice(&[0x08, 0xa1, 0x01]);
282        expected.extend_from_slice(&unhex(WRAP_COSE));
283        assert_eq!(without[0], 0xa3);
284        assert_eq!(with, expected);
285    }
