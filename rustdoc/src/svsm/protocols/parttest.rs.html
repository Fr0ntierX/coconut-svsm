5// Partition-isolation test protocol (F0X-78, idea B).
6//
7// B isolates partitions by separate page tables inside one VMPL, and rests on
8// one invariant: the SVSM owns a partition's page tables and the partition
9// cannot write its own page-table pages. That is RMP write-protection enforced
10// against the partition's VMPL, and it is the half of the invariant that cannot
11// be tested from inside the guest.
12//
13// The guest side is already settled on silicon: a partition at VMPL2 cannot
14// RMPADJUST VMPL2, VMPL1 or VMPL0 (all FAIL_PERMISSION), so it cannot re-grant
15// itself access the SVSM revoked. What was missing is proof that the revocation
16// is actually enforced when the partition then touches the page. This protocol
17// supplies the VMPL0 half so a guest prober can complete the test:
18//
19//   1. guest allocates a page it owns and writes to it              (succeeds)
20//   2. guest calls REVOKE_WRITE with that GPA
21//   3. guest writes again                                           (must fault)
22//   4. guest calls GRANT_WRITE to restore, so the page is reusable
23//
24// Read is deliberately left in place by REVOKE_WRITE rather than dropping all
25// access. Page tables have to stay readable for the hardware walker at the
26// partition's VMPL, so read-only is the configuration B actually runs, and
27// no-access would test a stronger property than the design needs.
28//
29// This is a test surface that can hand a guest a way to make its own pages
30// unwritable, so it is behind the `parttest` cargo feature and must never be
31// enabled in a shipping build. It is not registered in the protocol dispatch
32// unless that feature is on.

60/// Drop write for the guest VMPL on one 4K page, leaving read in place.
61const SVSM_REQ_PARTTEST_REVOKE_WRITE: u32 = 0;
62/// Restore read/write/execute for the guest VMPL on one 4K page.
63const SVSM_REQ_PARTTEST_GRANT_WRITE: u32 = 1;
64/// Do nothing and return success. Times the guest -> VMGEXIT -> hypervisor ->
65/// VMPL0 -> back round trip with no work inside it, which is the relay cost any
66/// hypervisor-mediated partition switch has to pay (F0X-93 vmsa_reentry). VEIL
67/// reports 7,135 cycles for a relayed domain switch and NestedSGX 32-34k, so
68/// this is the term to compare against those.
69const SVSM_REQ_PARTTEST_NOOP: u32 = 2;
70/// Drop every permission for the guest VMPL on one 4K page, read included.
71/// REVOKE_WRITE keeps read because B runs read-only: page tables must stay
72/// readable for the hardware walker at the partition's VMPL. This exists only
73/// to tell whether the wedged-vCPU behaviour is specific to read-only or is
74/// general to any RMP permission violation (F0X-301, experiment 2).
75const SVSM_REQ_PARTTEST_REVOKE_ALL: u32 = 3;
76/// Ask the PSP for a VMPL0 derived key and return the first 24 bytes of its
77/// SHA-384 in rcx, rdx and r8. Input: rcx = GUEST_FIELD_SELECT, rdx =
78/// ROOT_KEY_SELECT. This exists to check on silicon that the key the DICE
79/// root will use (F0X-80) is stable for one image and changes with the
80/// selected fields. The key itself is the monitor's root secret and never
81/// leaves VMPL0; only a one-way hash does, and only in parttest builds.
82const SVSM_REQ_PARTTEST_DERIVED_KEY_HASH: u32 = 4;
83/// Write the monitor's hardware root endorsement (F0X-80) to the 4K guest
84/// page at rcx: the alias public key (97-byte SEC1 point) followed by the
85/// VMPL0 attestation report that binds it in REPORT_DATA. A verifier checks
86/// the report against the VCEK chain and recomputes REPORT_DATA from the key.
87/// Only public values leave VMPL0; the CDI and the alias private key do not.
88#[cfg(feature = "dice")]
89const SVSM_REQ_PARTTEST_DICE_ROOT: u32 = 5;
90/// Measure a guest memory region as a partition and return its EAT
91/// (F0X-80): the monitor token, signed by the hardware-rooted monitor alias,
92/// carrying the partition's nested token and the monitor's endorsement of the
93/// partition's measurement and alias key. With call 5's root it is a complete
94/// version 2 evidence bundle.
95///
96/// rcx is a 4K request page: nonce (32 bytes), REPORT_DATA (64), region GPA
97/// (u64), region page count (u32, at most `DICE_ATTEST_MAX_PAGES`), partition
98/// ID length (u32, at most `DICE_ATTEST_MAX_ID`), then the ID in UTF-8. rdx
99/// is a 4K output page that gets the token length (u32) followed by the token.
100///
101/// r8 = flags, 0 for none. With `DICE_ATTEST_WRAP_KEY` the request page also
102/// holds the partition's wrap public key (F0X-898), a 97-byte uncompressed
103/// SEC1 P-384 point at `DICE_ATTEST_WRAP_OFFSET`, and the nested token carries
104/// it as a COSE_Key in its `cnf` claim. The point must be on the curve. A
105/// verifier also requires REPORT_DATA to be SHA-512(nonce || that COSE_Key).
106#[cfg(feature = "dice")]
107const SVSM_REQ_PARTTEST_DICE_ATTEST: u32 = 6;
108/// Retire a guest vCPU and revoke a partition's pages (F0X-75 watchdog
109/// reclaim), through `core::retire_guest_vcpu`. The guest must take the
110/// target CPU offline first.
111///
112/// In: rcx = the target's VMSA GPA. rdx = GPA of a 4K page list, or 0 for
113/// none: a u64 count (at most `RETIRE_MAX_PAGES`) followed by that many 4K
114/// GPAs. r8 bits 0-62 = FAIL_INUSE deadline in TSC ticks (0 for the
115/// default), bit 63 = retire even if the host refuses AP_DESTROY.
116///
117/// Out, on success and on every error: rcx = AP_DESTROY ticks, rdx = ticks
118/// from the first RMPADJUST on the VMSA to success, r8 = pages revoked
119/// (bits 0-15) | FAIL_INUSE retries (16-47) | AP_DESTROY refused (63).
120///
121/// Errors: INVALID_PARAMETER (not a guest-created in-use VMSA, the caller's
122/// own vCPU, or a bad list), INVALID_ADDRESS (a list entry is not a guest
123/// page or is a VMSA or calling area, entry index in rdx), BUSY (FAIL_INUSE
124/// past the deadline), and the protocol codes below, whose detail goes in
125/// rdx. See `RetireError` for what each leaves behind.
126const SVSM_REQ_PARTTEST_RETIRE_VCPU: u32 = 7;
127const RETIRE_MAX_PAGES: usize = PAGE_SIZE / 8 - 1;
128/// Host refused AP_DESTROY, nothing revoked. rdx = SW_EXITINFO1.
129const RETIRE_ERR_AP_DESTROY: u64 = 0x100;
130/// RMPADJUST on the VMSA failed, nothing revoked. rdx = status.
131const RETIRE_ERR_VMSA_ADJUST: u64 = 0x101;
132/// Retired, but revoking a partition page failed. rdx = index | status << 32.
133const RETIRE_ERR_PAGE_REVOKE: u64 = 0x102;
134/// Retired and pages revoked, but the scrubbed VMSA page stays revoked from
135/// the guest. rdx = status.
136const RETIRE_ERR_VMSA_RELEASE: u64 = 0x103;
137/// The VMSA page could not be mapped. The host was already asked to halt it.
138const RETIRE_ERR_NO_MAPPING: u64 = 0x104;
139/// Start a guest-built VMSA on another vCPU (F0X-75 watchdog tests). In: rcx
140/// = VMSA GPA, rdx = calling area GPA, r8 = APIC ID, as CREATE_VCPU. Runs
141/// CREATE_VCPU's checks and registration, then sends GHCB AP_CREATE at the
142/// guest VMPL from VMPL0, so the guest kernel needs no GHCB access of its own
143/// to hand a vCPU it took offline to a partition payload. If AP_CREATE fails
144/// (`START_ERR_AP_CREATE`) the VMSA stays registered but never ran, and call
145/// 7 retires it.
146const SVSM_REQ_PARTTEST_START_VCPU: u32 = 8;
147/// Arm a watchdog slot (`watchdog::watchdog_arm`). rcx = 4K request page:
148/// VMSA GPA, heartbeat GPA (8-byte aligned), deadline in TSC ticks, page
149/// count (at most `WATCHDOG_MAX_PAGES`), then that many 4K GPAs to revoke on
150/// reclaim, all u64. rdx = retry policy (`watchdog::RetryPolicy`): 0 for the
151/// default, or `ARM_POLICY_SET` with bits 0-15 = plain retries and
152/// `ARM_POLICY_NO_FORCE` to never force. r8 = with `ARM_POLICY_SET`, the
153/// retry window in TSC ticks (0 for none). Out: rcx = slot index.
154const SVSM_REQ_PARTTEST_WATCHDOG_ARM: u32 = 9;
155/// One watchdog period (`watchdog::watchdog_tick`). Only the first vCPU that
156/// ticks may tick again. Out: rcx = armed slots checked, rdx = slots whose
157/// vCPU was retired in this tick (bitmask), r8 = VMPL0 ticks spent (bits
158/// 0-31) | slots that finished with an error (bitmask) << 32 | slots whose
159/// retire failed and is retried next tick (bitmask) << 48. A slot retired
160/// with a clean-up error (`ReclaimedPartly`) is in both of the first two.
161const SVSM_REQ_PARTTEST_WATCHDOG_TICK: u32 = 10;
162/// Write one slot's record to a 4K guest page. In: rcx = slot, rdx = page.
163/// The record is `WATCHDOG_STATUS_WORDS` u64s, laid out in `watchdog_status`.
164const SVSM_REQ_PARTTEST_WATCHDOG_STATUS: u32 = 11;
165/// Free a slot that is armed, reclaimed or failed. In: rcx = slot.
166const SVSM_REQ_PARTTEST_WATCHDOG_DISARM: u32 = 12;
167/// Arm one injected failure at a named step of the retire, START_VCPU or
168/// watchdog paths (`faultinj`), so their error paths can be driven on
169/// hardware. In: rcx bits 0-11 = `FaultStep` code, or 0 to clear, bits 12-63
170/// = a page-aligned VMSA GPA the fault is keyed on (0 for any target; only
171/// the retire steps name a target). rdx = the status the
172/// failing step returns, where it returns one (for FAIL_INUSE, how many fake
173/// results the retire sees, u64::MAX for all of them). r8 bits 0-31 = hits at the
174/// step to let through first, bits 32-63 = how many hits fail (0 means 1).
175/// Out: rcx = the step of the fault this replaces (0 if none), rdx = how
176/// many times it fired, r8 = failures it had left.
177const SVSM_REQ_PARTTEST_FAULT_INJECT: u32 = 13;
178/// AP_CREATE refused by the host. rdx = SW_EXITINFO1, or u64::MAX if the GHCB
179/// answer was malformed.
180const START_ERR_AP_CREATE: u64 = 0x110;
181/// The tick came from a vCPU other than the watchdog vCPU.
182const WATCHDOG_ERR_NOT_WATCHDOG_CPU: u64 = 0x120;
183/// No free slot.
184const WATCHDOG_ERR_FULL: u64 = 0x121;
185/// The VMSA is already watched.
186const WATCHDOG_ERR_ALREADY_WATCHED: u64 = 0x122;
187const WATCHDOG_STATUS_WORDS: usize = 26;
188/// ARM rdx: a retry policy is given (otherwise the default).
189const ARM_POLICY_SET: u64 = 1 << 63;
190/// ARM rdx: never force, end `Failed` when the plain retries are used up.
191const ARM_POLICY_NO_FORCE: u64 = 1 << 62;
192#[cfg(feature = "dice")]
193const DICE_ATTEST_MAX_PAGES: usize = 256;
194#[cfg(feature = "dice")]
195const DICE_ATTEST_MAX_ID: usize = 64;
196/// Call 6 r8 flag: the request carries a wrap key for `cnf`.
197#[cfg(feature = "dice")]
198const DICE_ATTEST_WRAP_KEY: u64 = 1;
199/// Where the wrap key sits in the call 6 request page, after the longest ID.
200#[cfg(feature = "dice")]
201const DICE_ATTEST_WRAP_OFFSET: usize = 112 + DICE_ATTEST_MAX_ID;

203#[cfg(feature = "dice")]
204fn dice_root(params: &RequestParams) -> Result<(), SvsmReqError> {
205    use crate::dice::{ALIAS_PUBLIC_SIZE, root_endorsement};
206    use crate::greq::pld_report::AttestationReport;
207
208    let paddr = PhysAddr::from(params.rcx);
209    if !paddr.is_page_aligned() {
210        return Err(SvsmReqError::invalid_parameter());
211    }
212    let (public, report) = root_endorsement().map_err(|e| {
213        log::info!("parttest: DICE root endorsement failed: {e:?}");
214        SvsmReqError::invalid_request()
215    })?;
216    // guest_ptr checks the region is guest memory before anything is written.
217    let guard = PerCPUPageMappingGuard::create_4k(paddr)?;
218    guard
219        .guest_ptr::<[u8; ALIAS_PUBLIC_SIZE]>(0)?
220        .write(public)?;
221    guard
222        .guest_ptr::<AttestationReport>(ALIAS_PUBLIC_SIZE)?
223        .write(report)?;
224    Ok(())
225}

227#[cfg(feature = "dice")]
228fn dice_attest(params: &RequestParams) -> Result<(), SvsmReqError> {
229    use crate::dice::{WRAP_PUBLIC_SIZE, attest_partition, check_wrap_key};
230
231    let req_pa = PhysAddr::from(params.rcx);
232    let out_pa = PhysAddr::from(params.rdx);
233    if !req_pa.is_page_aligned() || !out_pa.is_page_aligned() {
234        return Err(SvsmReqError::invalid_parameter());
235    }
236    if params.r8 & !DICE_ATTEST_WRAP_KEY != 0 {
237        return Err(SvsmReqError::invalid_parameter());
238    }
239    let with_wrap_key = params.r8 & DICE_ATTEST_WRAP_KEY != 0;
240
241    let (nonce, report_data, region_pa, pages, id, wrap_key) = {
242        let guard = PerCPUPageMappingGuard::create_4k(req_pa)?;
243        let nonce = guard.guest_ptr::<[u8; 32]>(0)?.read()?;
244        let report_data = guard.guest_ptr::<[u8; 64]>(32)?.read()?;
245        let region_pa = guard.guest_ptr::<u64>(96)?.read()?;
246        let pages = guard.guest_ptr::<u32>(104)?.read()? as usize;
247        let id_len = guard.guest_ptr::<u32>(108)?.read()? as usize;
248        if id_len == 0 || id_len > DICE_ATTEST_MAX_ID {
249            return Err(SvsmReqError::invalid_parameter());
250        }
251        let id = guard.guest_slice::<u8>(112, id_len)?.read_to_vec()?;
252        let wrap_key = if with_wrap_key {
253            Some(
254                guard
255                    .guest_ptr::<[u8; WRAP_PUBLIC_SIZE]>(DICE_ATTEST_WRAP_OFFSET)?
256                    .read()?,
257            )
258        } else {
259            None
260        };
261        (
262            nonce,
263            report_data,
264            PhysAddr::from(region_pa),
265            pages,
266            id,
267            wrap_key,
268        )
269    };
270    let id = core::str::from_utf8(&id).map_err(|_| SvsmReqError::invalid_parameter())?;
271    // A wrap key that is not a P-384 point is the caller's mistake.
272    if let Some(w) = &wrap_key {
273        check_wrap_key(w).map_err(|_| SvsmReqError::invalid_parameter())?;
274    }
275    if !region_pa.is_page_aligned() || pages == 0 || pages > DICE_ATTEST_MAX_PAGES {
276        return Err(SvsmReqError::invalid_parameter());
277    }
278
279    // The partition measurement is SHA-384 over the region, page by page.
280    let mut hash = Sha384::new();
281    for i in 0..pages {
282        let pa = region_pa
283            .checked_add(i * PAGE_SIZE)
284            .ok_or_else(SvsmReqError::invalid_address)?;
285        let guard = PerCPUPageMappingGuard::create_4k(pa)?;
286        hash.update(guard.guest_slice::<u8>(0, PAGE_SIZE)?.read_to_vec()?);
287    }
288    let measurement: [u8; 48] = hash.finalize().into();
289
290    let token = attest_partition(&nonce, &report_data, id, &measurement, wrap_key.as_ref())
291        .map_err(|e| {
292            log::info!("parttest: DICE attest failed: {e:?}");
293            SvsmReqError::invalid_request()
294        })?;
295    let len = u32::try_from(token.len()).map_err(|_| SvsmReqError::invalid_request())?;
296    if token.len() > PAGE_SIZE - 4 {
297        return Err(SvsmReqError::invalid_request());
298    }
299    let guard = PerCPUPageMappingGuard::create_4k(out_pa)?;
300    guard.guest_ptr::<u32>(0)?.write(len)?;
301    guard
302        .guest_slice::<u8>(4, token.len())?
303        .write_from_slice(&token)?;
304    Ok(())
305}

310/// Maps the guest page named by rcx and applies `flags` to it for the guest
311/// VMPL. The address must be 4K aligned and inside a valid guest physical
312/// region, which is the same check the core PVALIDATE handler applies, so a
313/// guest cannot point this at SVSM-private or hypervisor memory.
314fn adjust_guest_page(params: &RequestParams, flags: RMPFlags) -> Result<(), SvsmReqError> {
315    let paddr = PhysAddr::from(params.rcx);
316
317    if !paddr.is_page_aligned() {
318        return Err(SvsmReqError::invalid_parameter());
319    }
320
321    let region =
322        MemoryRegion::checked_new(paddr, PAGE_SIZE).ok_or_else(SvsmReqError::invalid_address)?;
323    if !valid_phys_region(&region) {
324        return Err(SvsmReqError::invalid_address());
325    }
326
327    // Serialize against PVALIDATE and CREATE_VCPU, which change RMP state on
328    // guest pages under this lock. Without it a concurrent CREATE_VCPU on the
329    // same page could see its permissions change between its own steps.
330    let _lock = PVALIDATE_LOCK.lock_write();
331
332    let guard = PerCPUPageMappingGuard::create_4k(paddr)?;
333    let vaddr = guard.virt_addr();
334
335    // SAFETY: the page is a guest page, mapped for the duration of the guard,
336    // and the flags only ever narrow or restore the guest VMPL's own access to
337    // it. Nothing here touches SVSM-private mappings, so it cannot affect
338    // memory safety on this side.
339    // The RMPADJUST status code is kept in the log line rather than folded
340    // into the protocol error, which has no field for it. A failed lab run
341    // otherwise says only "invalid request" when the interesting part is
342    // whether the instruction returned FAIL_PERMISSION, FAIL_SIZEMISMATCH or
343    // FAIL_INUSE.
344    unsafe { rmp_adjust(vaddr, flags, PageSize::Regular) }.map_err(|e| match e {
345        SvsmError::SevSnp(code) => {
346            log::info!("parttest: RMPADJUST failed on {paddr:#x}, status {code:?}");
347            SvsmReqError::invalid_request()
348        }
349        other => other.into(),
350    })
351}

353fn derived_key_hash(params: &mut RequestParams) -> Result<(), SvsmReqError> {
354    let root_key_select =
355        u32::try_from(params.rdx).map_err(|_| SvsmReqError::invalid_parameter())?;
356    let request = SnpDerivedKeyRequest::new(root_key_select, params.rcx, 0, 0)
357        .map_err(|_| SvsmReqError::invalid_parameter())?;
358    let key = get_derived_key(&request).map_err(|e| {
359        log::info!("parttest: MSG_KEY_REQ failed: {e:?}");
360        SvsmReqError::invalid_request()
361    })?;
362    let digest = Sha384::digest(&key[..]);
363    let word = |i: usize| {
364        u64::from_le_bytes(
365            digest[i * 8..i * 8 + 8]
366                .try_into()
367                .expect("an 8-byte slice of a 48-byte digest"),
368        )
369    };
370    params.rcx = word(0);
371    params.rdx = word(1);
372    params.r8 = word(2);
373    Ok(())
374}

376fn retire_vcpu(params: &mut RequestParams) -> Result<(), SvsmReqError> {
377    let vmsa_pa = PhysAddr::from(params.rcx);
378    let list_pa = PhysAddr::from(params.rdx);
379    let deadline = params.r8 & !(1u64 << 63);
380    let opts = RetireOptions {
381        inuse_deadline: if deadline == 0 {
382            RETIRE_DEFAULT_INUSE_DEADLINE
383        } else {
384            deadline
385        },
386        force: params.r8 & (1u64 << 63) != 0,
387        log: true,
388    };
389    params.rcx = 0;
390    params.rdx = 0;
391    params.r8 = 0;
392
393    let mut pages: Vec<PhysAddr> = Vec::new();
394    if !list_pa.is_null() {
395        if !list_pa.is_page_aligned() {
396            return Err(SvsmReqError::invalid_parameter());
397        }
398        let guard = PerCPUPageMappingGuard::create_4k(list_pa)?;
399        let count = guard.guest_ptr::<u64>(0)?.read()?;
400        let count = usize::try_from(count).map_err(|_| SvsmReqError::invalid_parameter())?;
401        if count > RETIRE_MAX_PAGES {
402            return Err(SvsmReqError::invalid_parameter());
403        }
404        let entries = guard.guest_slice::<u64>(8, count)?.read_to_vec()?;
405        pages
406            .try_reserve_exact(count)
407            .map_err(|_| SvsmReqError::invalid_request())?;
408        pages.extend(entries.into_iter().map(PhysAddr::from));
409    }
410
411    let (res, stats) = match retire_guest_vcpu(vmsa_pa, &pages, &opts) {
412        Ok(stats) => (Ok(()), stats),
413        Err((e, stats)) => (Err(e), stats),
414    };
415    params.rcx = stats.ap_destroy_ticks;
416    params.rdx = stats.vmsa_ticks;
417    params.r8 = (stats.pages_revoked as u64 & 0xffff)
418        | (stats.inuse_retries & 0xffff_ffff) << 16
419        | u64::from(stats.ap_destroy_refused) << 63;
420
421    res.map_err(|e| {
422        log::info!("parttest: retire of VMSA {vmsa_pa:#x} failed: {e:?}");
423        match e {
424            RetireError::InvalidTarget => SvsmReqError::invalid_parameter(),
425            RetireError::InvalidPage(i) => {
426                params.rdx = i as u64;
427                SvsmReqError::invalid_address()
428            }
429            RetireError::NoMapping => SvsmReqError::protocol(RETIRE_ERR_NO_MAPPING),
430            RetireError::ApDestroy(code) => {
431                params.rdx = code;
432                SvsmReqError::protocol(RETIRE_ERR_AP_DESTROY)
433            }
434            RetireError::Busy(_) => SvsmReqError::busy(),
435            RetireError::VmsaAdjust(status) => {
436                params.rdx = status;
437                SvsmReqError::protocol(RETIRE_ERR_VMSA_ADJUST)
438            }
439            RetireError::PageRevoke { index, status } => {
440                params.rdx = index as u64 | status << 32;
441                SvsmReqError::protocol(RETIRE_ERR_PAGE_REVOKE)
442            }
443            RetireError::VmsaRelease(status) => {
444                params.rdx = status;
445                SvsmReqError::protocol(RETIRE_ERR_VMSA_RELEASE)
446            }
447        }
448    })
449}

451fn start_vcpu(params: &mut RequestParams) -> Result<(), SvsmReqError> {
452    let vmsa_pa = PhysAddr::from(params.rcx);
453    let apic_id = params.r8 & 0xffff_ffff;
454    core_create_vcpu(params)?;
455    let vmpl = fault_inject!(ApCreate, _ => FAULT_BAD_VMPL, RMPFlags::GUEST_VMPL.bits());
456    let sev_features = params.sev_features;
457    with_current_ghcb(|ghcb| ghcb.ap_create(vmsa_pa, apic_id, vmpl, sev_features)).map_err(|e| {
458        log::info!("parttest: AP_CREATE for APIC {apic_id} VMSA {vmsa_pa:#x} failed: {e:?}");
459        // The documented detail, as call 7 reports a refused AP_DESTROY.
460        params.rdx = match e {
461            SvsmError::Ghcb(GhcbError::VmgexitError(info1, _)) => info1,
462            _ => u64::MAX,
463        };
464        SvsmReqError::protocol(START_ERR_AP_CREATE)
465    })
466}
467
468fn fault_inject_call(params: &mut RequestParams) -> Result<(), SvsmReqError> {
469    let fault = match params.rcx & 0xfff {
470        0 => None,
471        code => {
472            let step = FaultStep::from_code(code).ok_or_else(SvsmReqError::invalid_parameter)?;
473            let count = (params.r8 >> 32) as u32;
474            Some(Fault {
475                step,
476                value: params.rdx,
477                skip: params.r8 as u32,
478                count: count.max(1),
479                key: params.rcx & !0xfff,
480            })
481        }
482    };
483    let prev = fault_arm(fault);
484    if let Some(f) = fault {
485        log::info!(
486            "parttest: fault armed at {:?}, value {:#x}, skip {}, count {}, key {:#x}",
487            f.step,
488            f.value,
489            f.skip,
490            f.count,
491            f.key
492        );
493    }
494    params.rcx = prev.step;
495    params.rdx = prev.fired;
496    params.r8 = u64::from(prev.remaining);
497    Ok(())
498}
499
500fn watchdog_err(e: WatchdogError, params: &mut RequestParams) -> SvsmReqError {
501    match e {
502        WatchdogError::BadPage(i) => {
503            params.rdx = i as u64;
504            SvsmReqError::invalid_address()
505        }
506        WatchdogError::BadHeartbeat => SvsmReqError::invalid_address(),
507        WatchdogError::Read(err) => {
508            log::info!("parttest: watchdog heartbeat read failed: {err:?}");
509            SvsmReqError::invalid_address()
510        }
511        WatchdogError::BadTarget | WatchdogError::TooManyPages | WatchdogError::BadSlot => {
512            SvsmReqError::invalid_parameter()
513        }
514        WatchdogError::AlreadyWatched => SvsmReqError::protocol(WATCHDOG_ERR_ALREADY_WATCHED),
515        WatchdogError::Full => SvsmReqError::protocol(WATCHDOG_ERR_FULL),
516        WatchdogError::NotWatchdogCpu => SvsmReqError::protocol(WATCHDOG_ERR_NOT_WATCHDOG_CPU),
517    }
518}
519
520fn watchdog_arm_call(params: &mut RequestParams) -> Result<(), SvsmReqError> {
521    let req_pa = PhysAddr::from(params.rcx);
522    if !req_pa.is_page_aligned() {
523        return Err(SvsmReqError::invalid_parameter());
524    }
525    let (vmsa_pa, hb_pa, deadline, pages) = {
526        let guard = PerCPUPageMappingGuard::create_4k(req_pa)?;
527        let head = guard.guest_slice::<u64>(0, 4)?.read_to_vec()?;
528        let count = usize::try_from(head[3]).map_err(|_| SvsmReqError::invalid_parameter())?;
529        if count > WATCHDOG_MAX_PAGES {
530            return Err(SvsmReqError::invalid_parameter());
531        }
532        let pages = guard.guest_slice::<u64>(32, count)?.read_to_vec()?;
533        (head[0], head[1], head[2], pages)
534    };
535    let pages: Vec<PhysAddr> = pages.into_iter().map(PhysAddr::from).collect();
536    let policy = if params.rdx & ARM_POLICY_SET != 0 {
537        RetryPolicy {
538            retries: params.rdx as u16,
539            window: params.r8,
540            force: params.rdx & ARM_POLICY_NO_FORCE == 0,
541        }
542    } else {
543        RetryPolicy::default()
544    };
545    params.rdx = 0;
546    let slot = watchdog_arm(
547        PhysAddr::from(vmsa_pa),
548        PhysAddr::from(hb_pa),
549        deadline,
550        pages,
551        policy,
552    )
553    .map_err(|e| watchdog_err(e, params))?;
554    params.rcx = slot as u64;
555    Ok(())
556}
557
558fn watchdog_tick_call(params: &mut RequestParams) -> Result<(), SvsmReqError> {
559    let report = watchdog_tick().map_err(|e| watchdog_err(e, params))?;
560    params.rcx = u64::from(report.checked);
561    params.rdx = report.reclaimed;
562    params.r8 = (report.ticks & 0xffff_ffff) | report.failed << 32 | report.retrying << 48;
563    Ok(())
564}

566/// A retire error as status words carry it: code << 8 | detail << 32.
567fn retire_error_code(e: RetireError) -> u64 {
568    let (code, detail): (u64, u64) = match e {
569        RetireError::InvalidTarget => (1, 0),
570        RetireError::InvalidPage(i) => (2, i as u64),
571        RetireError::NoMapping => (3, 0),
572        RetireError::ApDestroy(c) => (4, c),
573        RetireError::Busy(r) => (5, r),
574        RetireError::VmsaAdjust(s) => (6, s),
575        RetireError::PageRevoke { index, status } => (7, index as u64 | status << 16),
576        RetireError::VmsaRelease(s) => (8, s),
577    };
578    code << 8 | (detail & 0xffff_ffff) << 32
579}
580
581/// Bits 0-7 the state, and for the states that carry an error, that error as
582/// `retire_error_code` lays it out. Retrying sets bit 16 when the next
583/// attempt is forced.
584fn slot_state_code(state: SlotState, force_next: bool) -> u64 {
585    match state {
586        SlotState::Free => 0,
587        SlotState::Armed => 1,
588        SlotState::Reclaiming => 2,
589        SlotState::Reclaimed => 3,
590        SlotState::Failed(e) => 4 | retire_error_code(e),
591        SlotState::Retrying(e) => 5 | retire_error_code(e) | u64::from(force_next) << 16,
592        SlotState::ReclaimedPartly(e) => 6 | retire_error_code(e),
593    }
594}
595
596fn watchdog_status_call(params: &mut RequestParams) -> Result<(), SvsmReqError> {
597    let index = usize::try_from(params.rcx).map_err(|_| SvsmReqError::invalid_parameter())?;
598    let out_pa = PhysAddr::from(params.rdx);
599    if !out_pa.is_page_aligned() {
600        return Err(SvsmReqError::invalid_parameter());
601    }
602    let s = watchdog_status(index).map_err(|e| watchdog_err(e, params))?;
603    let words: [u64; WATCHDOG_STATUS_WORDS] = [
604        slot_state_code(s.state, s.force_next),
605        u64::from(s.vmsa_pa),
606        s.cpu_index as u64,
607        s.deadline,
608        s.last_value,
609        s.last_change,
610        s.armed_at,
611        s.changes_seen,
612        s.max_gap,
613        s.read_errors,
614        s.reclaim_decided,
615        s.stale_at_reclaim,
616        s.retire.ap_destroy_ticks,
617        s.retire.lock_ticks,
618        s.retire.vmsa_ticks,
619        s.retire.total_ticks,
620        s.retire.pages_revoked as u64 | s.retire.inuse_retries << 16,
621        s.watchdog_ticks,
622        s.last_tick,
623        rdtsc(),
624        s.retire_started,
625        s.max_still,
626        // 22: retries | an attempt used force << 32 | the host refused
627        // AP_DESTROY on the last attempt << 33.
628        u64::from(s.retries)
629            | u64::from(s.forced) << 32
630            | u64::from(s.retire.ap_destroy_refused) << 33,
631        // 23: the policy as ARM takes it.
632        ARM_POLICY_SET
633            | u64::from(s.policy.retries)
634            | if s.policy.force {
635                0
636            } else {
637                ARM_POLICY_NO_FORCE
638            },
639        // 24: retry window, TSC ticks.
640        s.policy.window,
641        // 25: first failed attempt's error, 0 if none.
642        s.first_error.map_or(0, retire_error_code),
643    ];
644    let guard = PerCPUPageMappingGuard::create_4k(out_pa)?;
645    guard
646        .guest_slice::<u64>(0, WATCHDOG_STATUS_WORDS)?
647        .write_from_slice(&words)?;
648    Ok(())
649}
650
651pub fn parttest_protocol_request(
652    request: u32,
653    params: &mut RequestParams,
654) -> Result<(), SvsmReqError> {
655    match request {
656        // These touch GUEST_VMPL alone, where upstream's
657        // rmp_revoke_guest_access sweeps GUEST_VMPL through VMPL3. Deliberate:
658        // the test is about what the partition's own VMPL can reach, and
659        // narrowing one level is what idea B does on a switch. Sweeping the
660        // rest would also revoke levels the test never granted, which would
661        // make a failure harder to attribute rather than easier.
662        SVSM_REQ_PARTTEST_REVOKE_WRITE => {
663            adjust_guest_page(params, RMPFlags::GUEST_VMPL | RMPFlags::READ)
664        }
665        SVSM_REQ_PARTTEST_GRANT_WRITE => {
666            adjust_guest_page(params, RMPFlags::GUEST_VMPL | RMPFlags::RWX)
667        }
668        SVSM_REQ_PARTTEST_REVOKE_ALL => {
669            adjust_guest_page(params, RMPFlags::GUEST_VMPL | RMPFlags::NONE)
670        }
671        SVSM_REQ_PARTTEST_NOOP => Ok(()),
672        SVSM_REQ_PARTTEST_DERIVED_KEY_HASH => derived_key_hash(params),
673        SVSM_REQ_PARTTEST_RETIRE_VCPU => retire_vcpu(params),
674        SVSM_REQ_PARTTEST_START_VCPU => start_vcpu(params),
675        SVSM_REQ_PARTTEST_WATCHDOG_ARM => watchdog_arm_call(params),
676        SVSM_REQ_PARTTEST_WATCHDOG_TICK => watchdog_tick_call(params),
677        SVSM_REQ_PARTTEST_WATCHDOG_STATUS => watchdog_status_call(params),
678        SVSM_REQ_PARTTEST_WATCHDOG_DISARM => {
679            let index =
680                usize::try_from(params.rcx).map_err(|_| SvsmReqError::invalid_parameter())?;
681            watchdog_disarm(index).map_err(|e| watchdog_err(e, params))
682        }
683        SVSM_REQ_PARTTEST_FAULT_INJECT => fault_inject_call(params),
684        #[cfg(feature = "dice")]
685        SVSM_REQ_PARTTEST_DICE_ROOT => dice_root(params),
686        #[cfg(feature = "dice")]
687        SVSM_REQ_PARTTEST_DICE_ATTEST => dice_attest(params),
688        _ => Err(SvsmReqError::unsupported_call()),
689    }
690}
