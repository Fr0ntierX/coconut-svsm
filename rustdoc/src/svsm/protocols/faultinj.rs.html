5// Fault injection for the retire and watchdog error paths (F0X-75).
6//
7// Most of those paths need the host to refuse a request or the hardware to
8// return a status it never returns on the lab part, so they cannot be reached
9// from a guest. This module lets a parttest guest arm one failure at a named
10// step, and the step's call site then fails the way the real failure would:
11//
12// * AP_DESTROY and AP_CREATE are sent to the real host with an invalid VMPL,
13//   which KVM refuses before it looks at the target. The SVSM's handling of
14//   the host's answer is exercised as it is.
15// * RMPADJUST failures and mapping failures replace the operation, so the RMP
16//   is left exactly as a real failure would leave it.
17//
18// A fault can be keyed on one target (the VMSA GPA for the retire steps). It
19// then fires only at call sites that name that target, so a test can fail one
20// slot's retires while the watchdog reclaims another slot in the same tick.
21//
22// The whole module, and every call site through `fault_inject!`, exists only
23// with the `parttest` feature. A build without it has no fault state and no
24// injected branches.
25
26use crate::locking::SpinLock;
27
28/// A VMPL KVM refuses in every AP Creation request (it checks the VMPL is
29/// below 4 before anything else). The GHCB helpers keep 4 bits of it.
30pub(crate) const FAULT_BAD_VMPL: u64 = 0xf;
31
32/// Where a failure can be injected.
33#[derive(Debug, Clone, Copy, PartialEq, Eq)]
34#[repr(u64)]
35pub(crate) enum FaultStep {
36    /// The retire's AP_DESTROY goes to the host with an invalid VMPL.
37    ApDestroy = 1,
38    /// Mapping the VMSA page in the retire fails.
39    VmsaMap = 2,
40    /// The retire this hit lands in sees the armed value's worth of
41    /// FAIL_INUSE results from the VMSA RMPADJUST, without it running, and
42    /// then the real one. u64::MAX keeps failing until the deadline. Every
43    /// retire is one hit, so `skip` picks the retire.
44    VmsaInuse = 3,
45    /// RMPADJUST on the VMSA returns the armed value without running.
46    VmsaAdjust = 4,
47    /// Revoking a partition page returns the armed value without running.
48    /// Every page is one hit, so `skip` picks the page index.
49    PageRevoke = 5,
50    /// The first RMPADJUST that hands the old VMSA page back (VMPL3 none)
51    /// returns the armed value without running.
52    ReleaseVmpl3 = 6,
53    /// The old VMSA page looks read-only in the host's mapping, so it is not
54    /// scrubbed.
55    ReleaseReadOnly = 7,
56    /// The RMPADJUST that gives the scrubbed VMSA page back to the guest VMPL
57    /// returns the armed value without running.
58    ReleaseGrant = 8,
59    /// START_VCPU's AP_CREATE goes to the host with an invalid VMPL.
60    ApCreate = 9,
61    /// A watchdog heartbeat read fails. Every read is one hit.
62    HeartbeatRead = 10,
63    /// Arming a watchdog slot finds no free slot.
64    WatchdogFull = 11,
65}
66
67impl FaultStep {
68    pub(crate) fn from_code(code: u64) -> Option<Self> {
69        Some(match code {
70            1 => Self::ApDestroy,
71            2 => Self::VmsaMap,
72            3 => Self::VmsaInuse,
73            4 => Self::VmsaAdjust,
74            5 => Self::PageRevoke,
75            6 => Self::ReleaseVmpl3,
76            7 => Self::ReleaseReadOnly,
77            8 => Self::ReleaseGrant,
78            9 => Self::ApCreate,
79            10 => Self::HeartbeatRead,
80            11 => Self::WatchdogFull,
81            _ => return None,
82        })
83    }
84}

86/// One armed failure.
87#[derive(Debug, Clone, Copy)]
88pub(crate) struct Fault {
89    pub step: FaultStep,
90    /// What the failing step returns, where it returns a status.
91    pub value: u64,
92    /// Hits at the step to let through before the first failure.
93    pub skip: u32,
94    /// Failures left. The fault disarms itself at zero.
95    pub count: u32,
96    /// Fire only at call sites keyed on this value (a page-aligned VMSA GPA
97    /// for the retire steps). 0 fires at every call site of the step, keyed
98    /// or not. Skips count matching hits only.
99    pub key: u64,
100}
101
102/// What `fault_arm` replaced, for the guest to confirm a fault fired.
103#[derive(Debug, Clone, Copy, Default)]
104pub(crate) struct FaultReport {
105    /// Step of the previous fault, 0 if none was ever armed.
106    pub step: u64,
107    /// Times it fired.
108    pub fired: u64,
109    /// Failures it had left.
110    pub remaining: u32,
111}
112
113struct FaultState {
114    armed: Option<Fault>,
115    last_step: u64,
116    fired: u64,
117}
118
119static FAULT: SpinLock<FaultState> = SpinLock::new(FaultState {
120    armed: None,
121    last_step: 0,
122    fired: 0,
123});
124
125/// Arm `fault`, or clear it with `None`, and report the one it replaces.
126pub(crate) fn fault_arm(fault: Option<Fault>) -> FaultReport {
127    let mut st = FAULT.lock();
128    let report = FaultReport {
129        step: st.last_step,
130        fired: st.fired,
131        remaining: st.armed.map_or(0, |f| f.count),
132    };
133    st.armed = fault.filter(|f| f.count > 0);
134    st.last_step = fault.map_or(0, |f| f.step as u64);
135    st.fired = 0;
136    report
137}
138
139/// Called at `step`. Returns the armed value when this hit should fail.
140pub(crate) fn fault_hit(step: FaultStep) -> Option<u64> {
141    fault_hit_at(step, 0)
142}
143
144/// Called at `step` working on `key`. A fault keyed on another value lets
145/// the hit through without counting it.
146pub(crate) fn fault_hit_at(step: FaultStep, key: u64) -> Option<u64> {
147    let mut st = FAULT.lock();
148    let f = st
149        .armed
150        .as_mut()
151        .filter(|f| f.step == step && (f.key == 0 || f.key == key))?;
152    if f.skip > 0 {
153        f.skip -= 1;
154        return None;
155    }
156    f.count -= 1;
157    let value = f.value;
158    if f.count == 0 {
159        st.armed = None;
160    }
161    st.fired += 1;
162    Some(value)
163}

165#[cfg(test)]
166mod tests {
167    use super::*;
168
169    // One test touches the global, so the cases cannot race each other.
170    #[test]
171    fn arm_skip_count_and_report() {
172        fault_arm(None);
173        assert_eq!(fault_hit(FaultStep::ApDestroy), None);
174
175        let prev = fault_arm(Some(Fault {
176            step: FaultStep::PageRevoke,
177            value: 6,
178            skip: 2,
179            count: 2,
180            key: 0,
181        }));
182        assert_eq!(prev.step, 0);
183        assert_eq!(fault_hit(FaultStep::VmsaAdjust), None);
184        assert_eq!(fault_hit(FaultStep::PageRevoke), None);
185        assert_eq!(fault_hit(FaultStep::PageRevoke), None);
186        assert_eq!(fault_hit(FaultStep::PageRevoke), Some(6));
187        assert_eq!(fault_hit(FaultStep::PageRevoke), Some(6));
188        assert_eq!(fault_hit(FaultStep::PageRevoke), None);
189
190        let prev = fault_arm(Some(Fault {
191            step: FaultStep::VmsaInuse,
192            value: 0,
193            skip: 0,
194            count: 5,
195            key: 0,
196        }));
197        assert_eq!(prev.step, FaultStep::PageRevoke as u64);
198        assert_eq!(prev.fired, 2);
199        assert_eq!(prev.remaining, 0);
200        assert_eq!(fault_hit(FaultStep::VmsaInuse), Some(0));
201
202        let prev = fault_arm(None);
203        assert_eq!(prev.fired, 1);
204        assert_eq!(prev.remaining, 4);
205        assert_eq!(fault_hit(FaultStep::VmsaInuse), None);
206        assert_eq!(FaultStep::from_code(12), None);
207
208        // Keyed: other keys and unkeyed sites pass and do not use the skip.
209        fault_arm(Some(Fault {
210            step: FaultStep::ApDestroy,
211            value: 0,
212            skip: 1,
213            count: 1,
214            key: 0x5000,
215        }));
216        assert_eq!(fault_hit_at(FaultStep::ApDestroy, 0x6000), None);
217        assert_eq!(fault_hit(FaultStep::ApDestroy), None);
218        assert_eq!(fault_hit_at(FaultStep::ApDestroy, 0x5000), None);
219        assert_eq!(fault_hit_at(FaultStep::ApDestroy, 0x6000), None);
220        assert_eq!(fault_hit_at(FaultStep::ApDestroy, 0x5000), Some(0));
221        assert_eq!(fault_arm(None).fired, 1);
222        assert_eq!(FaultStep::from_code(5), Some(FaultStep::PageRevoke));
223    }
