7//! API to send `SNP_GUEST_REQUEST` commands to the PSP

12use crate::greq::{

16    pld_report::{SnpReportRequest, SnpReportResponse},
17};
18use core::mem::size_of;
19use zerocopy::IntoBytes;
20use zeroize::Zeroizing;

22const REPORT_REQUEST_SIZE: usize = size_of::<SnpReportRequest>();
23const REPORT_RESPONSE_SIZE: usize = size_of::<SnpReportResponse>();
24
25fn get_report(buffer: &mut [u8], certs: Option<&mut [u8]>) -> Result<usize, SvsmError> {
26    let request: &SnpReportRequest = SnpReportRequest::try_from_as_ref(buffer)?;
27    // Non-VMPL0 attestation reports can be requested by the guest kernel
28    // directly to the PSP.
29    if !request.is_vmpl0() {
30        return Err(SvsmError::InvalidParameter);
31    }
32    let response_len = if certs.is_none() {
33        send_regular_guest_request(
34            SnpGuestRequestMsgType::ReportRequest,
35            buffer,
36            REPORT_REQUEST_SIZE,
37        )?
38    } else {
39        send_extended_guest_request(
40            SnpGuestRequestMsgType::ReportRequest,
41            buffer,
42            REPORT_REQUEST_SIZE,
43            certs.unwrap(),
44        )?
45    };
46    if REPORT_RESPONSE_SIZE > response_len {
47        return Err(SvsmError::InvalidParameter);
48    }
49    let (response, _rest) =
50        SnpReportResponse::ref_from_prefix(buffer).map_err(|_| SvsmError::InvalidParameter)?;
51    response.validate()?;
52
53    Ok(response_len)
54}

56/// Request a regular VMPL0 attestation report to the PSP.
57///
58/// Use the `SNP_GUEST_REQUEST` driver to send the provided `MSG_REPORT_REQ` command to
59/// the PSP. The VPML field of the command must be set to zero.
60///
61/// The VMPCK0 is disabled for subsequent calls if this function fails in a way that
62/// the VM state can be compromised.
63///
64/// # Arguments
65///
66/// * `buffer`: Buffer with the [`MSG_REPORT_REQ`](SnpReportRequest) command that will be
67///   sent to the PSP. It must be large enough to hold the
68///   [`MSG_REPORT_RESP`](SnpReportResponse) received from the PSP.
69///
70/// # Returns
71///
72/// * Success
73///     * `usize`: Number of bytes written to `buffer`. It should match the
74///       [`MSG_REPORT_RESP`](SnpReportResponse) size.
75/// * Error
76///     * [`SvsmError`]
77pub fn get_regular_report(buffer: &mut [u8]) -> Result<usize, SvsmError> {
78    get_report(buffer, None)
79}

81/// Request an extended VMPL0 attestation report to the PSP.
82///
83/// We say that it is extended because it requests a VMPL0 attestation report
84/// to the PSP (as in [`get_regular_report()`]) and also requests to the hypervisor
85/// the certificates required to verify the attestation report.
86///
87/// The VMPCK0 is disabled for subsequent calls if this function fails in a way that
88/// the VM state can be compromised.
89///
90/// # Arguments
91///
92/// * `buffer`: Buffer with the [`MSG_REPORT_REQ`](SnpReportRequest) command that will be
93///   sent to the PSP. It must be large enough to hold the
94///   [`MSG_REPORT_RESP`](SnpReportResponse) received from the PSP.
95/// * `certs`:  Buffer to store the SEV-SNP certificates received from the hypervisor.
96///
97/// # Return codes
98///
99/// * Success
100///     * `usize`: Number of bytes written to `buffer`. It should match
101///       the [`MSG_REPORT_RESP`](SnpReportResponse) size.
102/// * Error
103///     * [`SvsmError`]
104///     * `SvsmError::Ghcb(GhcbError::VmgexitError(certs_buffer_size, psp_rc))`:
105///         * `certs` is not large enough to hold the certificates.
106///             * `certs_buffer_size`: number of bytes required.
107///             * `psp_rc`: PSP return code
108pub fn get_extended_report(buffer: &mut [u8], certs: &mut [u8]) -> Result<usize, SvsmError> {
109    get_report(buffer, Some(certs))
110}

112/// Request a VMPL0 key derived by the PSP.
113///
114/// The key mixes a chip-unique root with VMPL0 and the guest fields the
115/// request selects, so it can only be reproduced by the same guest on the
116/// same chip. It is a secret: callers must not hand it to a lower VMPL.
117/// The working buffer is zeroized before returning; the returned key is
118/// zeroized when dropped.
119///
120/// # Returns
121///
122/// * Success: the 32-byte derived key
123/// * Error: [`SvsmError`], including `SnpGuestRequest(status)` if the PSP
124///   refused the request (0x16 for invalid parameters)
125pub fn get_derived_key(
126    request: &SnpDerivedKeyRequest,
127) -> Result<Zeroizing<[u8; DERIVED_KEY_SIZE]>, SvsmError> {
128    let mut buffer = Zeroizing::new([0u8; size_of::<SnpDerivedKeyResponse>()]);
129    buffer[..size_of::<SnpDerivedKeyRequest>()].copy_from_slice(request.as_bytes());
130    let response_len = send_regular_guest_request(
131        SnpGuestRequestMsgType::KeyRequest,
132        &mut buffer[..],
133        size_of::<SnpDerivedKeyRequest>(),
134    )?;
135    if response_len < size_of::<SnpDerivedKeyResponse>() {
136        return Err(SvsmError::InvalidParameter);
137    }
138    let (response, _rest) = SnpDerivedKeyResponse::ref_from_prefix(&buffer[..])
139        .map_err(|_| SvsmError::InvalidParameter)?;
140    Ok(Zeroizing::new(response.key()?))
141}

143#[cfg(test)]
144mod tests {
145    #[allow(unused)]
146    use super::*;
147
148    #[test]
149    #[cfg_attr(not(test_in_svsm), ignore = "Can only be run inside guest")]
150    #[cfg(test_in_svsm)]
151    fn test_snp_launch_measurement() {
152        extern crate alloc;
153
154        use crate::serial::Terminal;
155        use crate::testing::{IORequest, assert_eq_warn, svsm_test_io};
156        use crate::testutils::{has_test_iorequests, is_test_platform_type};
157
158        use alloc::vec;
159        use bootdefs::platform::SvsmPlatformType;
160
161        if has_test_iorequests() && is_test_platform_type(SvsmPlatformType::Snp) {
162            let sp = svsm_test_io().unwrap();
163
164            sp.put_byte(IORequest::GetLaunchMeasurement as u8);
165
166            let mut expected_measurement = [0u8; 48];
167            for byte in &mut expected_measurement {
168                *byte = sp.get_byte();
169            }
170
171            let mut buf = vec![0; size_of::<SnpReportResponse>()];
172            let size = get_regular_report(&mut buf).unwrap();
173            assert_eq!(size, buf.len());
174
175            let (response, _rest) = SnpReportResponse::ref_from_prefix(&buf).unwrap();
176            response.validate().unwrap();
177            // FIXME: we still have some cases where the precalculated value does
178            // not match, so for now we just issue a warning until we fix the problem.
179            assert_eq_warn!(expected_measurement, *response.measurement());
180        }
181    }
